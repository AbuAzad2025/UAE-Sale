--- products ---
  ck_product_stock_non_negative: current_stock >= 0
  ck_product_price_non_negative: regular_price >= 0
  ck_product_cost_non_negative: cost_price >= 0
--- sales ---
  ck_sale_total_non_negative: total_amount >= 0
  ck_sale_paid_non_negative: paid_amount_base >= 0
  ck_sale_balance_non_negative: balance_due >= 0
--- sale_lines ---
  ck_saleline_qty_positive: quantity > 0
  ck_saleline_price_non_negative: unit_price >= 0
  ck_saleline_total_non_negative: line_total >= 0
  ck_saleline_discount_range: discount_percent >= 0 AND discount_percent <= 100
--- purchases ---
  ck_purchase_total_non_negative: total_amount >= 0
  ck_purchase_amount_non_negative: amount_base >= 0
--- purchase_lines ---
  ck_purchaseline_qty_positive: quantity > 0
  ck_purchaseline_cost_non_negative: unit_cost >= 0
  ck_purchaseline_total_non_negative: line_total >= 0
--- payments ---
  ck_payment_amount_positive: amount > 0
  ck_payment_amount_base_positive: amount_base > 0
  ck_payment_direction_valid: direction IN ('incoming', 'outgoing')
--- receipts ---
  ck_receipt_amount_positive: amount > 0
  ck_receipt_amount_base_positive: amount_base > 0
  ck_receipt_xor_parent: (sale_id IS NOT NULL AND purchase_id IS NULL) OR (sale_id IS NULL AND purchase_id IS NOT NULL) OR (sale_id IS NULL AND purchase_id IS NULL)
